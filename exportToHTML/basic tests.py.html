import numpy as np
import torch
from DNN import DNN, PWLUDNN
from sigmoidNN import sigmoidNN
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import PiecewiseLinearActivation as pwlu

#create a simple step function to approximate with the DNN class
def step_function(x):
    return np.where(x > 0, 3.0, 0.0)

#generate some sample data
def generate_data(num_samples=100):
    x = np.linspace(-10, 10, num_samples).reshape(-1, 1)  # 100 samples from 0 to 10
    y = step_function(x)
    return torch.tensor(x, dtype=torch.float32), torch.tensor(y, dtype=torch.float32)




#train dnn with the generated data
def train_dnn(dnn, x, y, num_epochs=100, learning_rate=0.01):
    criterion = torch.nn.MSELoss()
    optimizer = torch.optim.Adam(dnn.parameters(), lr=learning_rate)

    for epoch in range(num_epochs):
        dnn.train()
        optimizer.zero_grad()
        outputs = dnn.forward(x)
        loss = criterion(outputs, y)
        loss.backward()
        optimizer.step()

        #decrease delta of all PiecewiseLinearActivation modules by 10% each epoch
        for module in dnn.modules():
            if isinstance(module, pwlu.PiecewiseLinearActivation):
                module.delta *= 0.99


        print(f'Epoch {epoch+1=}, Loss: {loss.item():.4f}')
        if epoch <= 10 or epoch % 100 == 0:
            print()
            plotNN(x, y, n=epoch/10)
    return dnn

#print parameters of the DNN
def print_dnn_parameters(dnn):
    for name, param in dnn.named_parameters():
        print(f'{name}: {param.data}')

def plotNN(x_samples, y_samples, n = None):
    with torch.no_grad():
        x_test = torch.linspace(-10, 10, 200).reshape(-1, 1)
        y_pred = dnn.forward(x_test)
    plt.plot(x_test.numpy(), y_pred.numpy(), color='blue', label='DNN Prediction')
    plt.scatter(x_samples.numpy(), y_samples.numpy(), color='red', label='Training Data')
    #if n is not None, add "n" to the title
    if n is not None:
        plt.title(f'DNN Approximation of Step Function (n={n})')
    else:
        plt.title('DNN Approximation of Step Function final')
    plt.xlabel('Input')
    plt.ylabel('Output')
    # show the plot
    plt.legend()
    plt.savefig(f'plot_{n}.png')
    plt.close()


#main function that runs the tests
if __name__ == "__main__":
    #dnn = sigmoidNN(input_size=1, hidden_layers=[1], output_size=1)

    dnn = PWLUDNN(input_size=1, hidden_layers=[2], output_size=1, discontinuous_indices=[0], STE_true=True, ReLU_true=True)
    dnn.load_state_dict(torch.load('dnn_recent.pth'))
    #freeze all parameters except first layer parameters
    for name, param in dnn.named_parameters():
        if 'discontinuous_0' not in name:
            param.requires_grad = True
        else:
            param.requires_grad = True


    x_samples, y_samples = generate_data(100)
    print_dnn_parameters(dnn)
    dnn = train_dnn(dnn, x_samples, y_samples, num_epochs=2000, learning_rate=0.01)

    #plot the dnn over the data
    dnn.eval()
    with torch.no_grad():
        x_test = torch.linspace(0, 10, 100).reshape(-1, 1)
