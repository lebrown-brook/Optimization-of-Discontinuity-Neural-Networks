import numpy as np
import torch
from collections import OrderedDict
import discontinuous_layers as dl
import PiecewiseLinearActivation as pwlu

#Neural network class with discontinuous layers at given indices inherits from torch.nn.Module
class DNN(torch.nn.Module):
    def __init__(self, input_size, hidden_layers, output_size, discontinuous_indices=None, epsilon_init = None, STE_true = False, ReLU_true = True):
        super(DNN, self).__init__()
        self.input_size = input_size
        self.hidden_layers = hidden_layers
        self.output_size = output_size
        self.discontinuous_indices = discontinuous_indices if discontinuous_indices is not None else []
        self.epsilon_init = epsilon_init
        self.STE_true = STE_true
        self.ReLU_true = ReLU_true


        self.layers = self.build_network()


    def build_network(self):
        layer_sizes = [self.input_size] + self.hidden_layers + [self.output_size]
        layers = OrderedDict()
        for i in range(len(layer_sizes) - 2):
            if i in self.discontinuous_indices:
                layer_name = f"discontinuous_{i}"
                layers[layer_name] = dl.DiscontinuousLayer(layer_sizes[i], layer_sizes[i + 1], epsilon_init=self.epsilon_init, STE_true = self.STE_true, ReLU_true=self.ReLU_true)
            else:
                layer_name = f"dense_{i}"
                layers[layer_name]= torch.nn.Linear(layer_sizes[i], layer_sizes[i + 1])
                layers[f"relu_{i}"] = torch.nn.ELU()

        layer_name = f"output_layer"
        layers[layer_name] = torch.nn.Linear(layer_sizes[-2], layer_sizes[-1])
        return torch.nn.Sequential(layers)

    def forward(self, x):
        return self.layers(x)

class PWLUDNN(torch.nn.Module):
    def __init__(self, input_size, hidden_layers, output_size, discontinuous_indices=None, epsilon_init = None, STE_true = False, ReLU_true = True):
        super(PWLUDNN, self).__init__()
        self.input_size = input_size
        self.hidden_layers = hidden_layers
        self.output_size = output_size
        self.discontinuous_indices = discontinuous_indices if discontinuous_indices is not None else []
        self.epsilon_init = epsilon_init
        self.STE_true = STE_true
        self.ReLU_true = ReLU_true


        self.layers = self.build_network()


    def build_network(self):
        layer_sizes = [self.input_size] + self.hidden_layers + [self.output_size]
        layers = OrderedDict()
        for i in range(len(layer_sizes) - 2):
            if i in self.discontinuous_indices:
                layer_name = f"discontinuous_{i}"
                layers[layer_name] = torch.nn.Linear(layer_sizes[i], layer_sizes[i + 1])
                layers[f"PWLU_discontinuous_{i}"] = pwlu.PiecewiseLinearActivation(layer_sizes[i + 1],
                                                                                   epsilon_init=self.epsilon_init)
            else:
                layer_name = f"dense_{i}"
                layers[layer_name]= torch.nn.Linear(layer_sizes[i], layer_sizes[i + 1])
                layers[f"relu_{i}"] = torch.nn.ELU()

        layer_name = f"output_layer"
        layers[layer_name] = torch.nn.Linear(layer_sizes[-2], layer_sizes[-1])
        return torch.nn.Sequential(layers)

    def forward(self, x):
        return self.layers(x)
