#differentiable heaviside function
class heaviside_function(torch.autograd.Function):
    @staticmethod
    def forward(ctx, input):
        return (input > 0).float()

    @staticmethod
    def backward(ctx, grad_output):
        # Subgradient is zero everywhere
        return torch.zeros_like(grad_output), None

# discontinuous layer which has a relu + epsilon * heaviside function
class DiscontinuousLayer(torch.nn.Module):
    def __init__(self, input_size, output_size, epsilon_init = None, STE_true = False, ReLU_true = True):
        super(DiscontinuousLayer, self).__init__()
        self.linear = torch.nn.Linear(input_size, output_size)
        self.relu = torch.nn.ELU()
        #init linear layer with 1s for weights and -5 for biases
            #self.linear.weight.data.fill_(1.0)
            #self.linear.bias.data.fill_(-5.0)
        if epsilon_init is not None:
            self.epsilon = torch.nn.Parameter(torch.tensor(epsilon_init, dtype=torch.float32, device='cuda' if torch.cuda.is_available() else 'cpu'))
        else:
            #init epsilon with random values
            self.epsilon = torch.nn.Parameter(torch.randn(output_size, dtype=torch.float32, device='cuda' if torch.cuda.is_available() else 'cpu'))

        self.STE_true = STE_true
        self.ReLU_true = ReLU_true

        #init linear layer weights and biases randomly
        torch.nn.init.xavier_uniform_(self.linear.weight)

    def forward(self, x):
        x = self.linear(x)
        if self.STE_true:
            y = torch.mul(self.epsilon, heaviside_function_STE.apply(x))
        else:
            y = torch.mul(self.epsilon, heaviside_function.apply(x))

        if self.ReLU_true:
            return self.relu(x) + y
        else:
            return y
