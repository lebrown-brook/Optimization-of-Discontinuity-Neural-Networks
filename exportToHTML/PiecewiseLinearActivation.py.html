import torch

class PiecewiseLinearActivationFunction(torch.autograd.Function):
    @staticmethod
    def forward(ctx, input, epsilon, delta):
        ctx.save_for_backward(input, epsilon)
        ctx.delta = delta

        case1 = torch.zeros_like(input)  # x < 0
        case2 = (epsilon / delta + 1) * input  # 0 <= x <= delta
        case3 = input + epsilon  # x > delta

        output = torch.where(
            input <= 0, case1,
            torch.where(input <= delta, case2, case3)
        )
        return output

    @staticmethod
    def backward(ctx, grad_output):
        input, epsilon = ctx.saved_tensors
        delta = ctx.delta

        # Derivatives for each region
        grad_input = torch.where(
            input <= 0, torch.zeros_like(input),
            torch.where(input <= delta, (epsilon / delta + 1), torch.ones_like(input))
        )

        grad_epsilon = torch.where(
            input <= 0, torch.zeros_like(input),
            torch.where(input <= delta, input / delta, torch.ones_like(input))
        )
        grad_epsilon = grad_output * grad_epsilon  # Element-wise
        # If epsilon is a vector, sum over batch dimension
        if grad_epsilon.shape == epsilon.shape:
            pass
        else:
            grad_epsilon = grad_epsilon.sum(dim=0)

        return grad_output * grad_input, grad_epsilon, None


class PiecewiseLinearActivation(torch.nn.Module):
    def __init__(self, output_size, epsilon_init=None, delta=0.1):
        super().__init__()
        self.delta = delta  # Not trainable

        if epsilon_init is not None:
            self.epsilon = torch.nn.Parameter(torch.tensor(epsilon_init, dtype=torch.float32, device='cuda' if torch.cuda.is_available() else 'cpu'))
        else:
            #init epsilon with random values
            self.epsilon = torch.nn.Parameter(torch.randn(output_size, dtype=torch.float32, device='cuda' if torch.cuda.is_available() else 'cpu'))

    def forward(self, x):
        return PiecewiseLinearActivationFunction.apply(x, self.epsilon, self.delta)
